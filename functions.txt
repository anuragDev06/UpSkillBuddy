    const handleCalculateMatch = (e) => {
        e.preventDefault();
        setMatchLoading(true);
        setMatchResult(null);

        setTimeout(() => {
            const resumeWords = resumeSkills.toLowerCase().split(/[,\s]+/);
            const reqSkills = ['react', 'javascript', 'tailwind', 'redux', 'typescript', 'jest', 'rest api'];
            const matched = [];
            const missing = [];

            reqSkills.forEach(skill => {
                if (resumeWords.some(w => w.includes(skill) || skill.includes(w))) {
                    matched.push(skill);
                } else {
                    missing.push(skill);
                }
            });

            const scorePercent = Math.round(50 + (matched.length / reqSkills.length) * 45);

            setMatchResult({
                score: scorePercent,
                matched: matched.map(s => s.charAt(0).toUpperCase() + s.slice(1)),
                missing: missing.map(s => s.charAt(0).toUpperCase() + s.slice(1)),
